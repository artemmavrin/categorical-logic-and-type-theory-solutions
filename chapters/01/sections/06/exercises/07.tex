\begin{exercise}
Let \(\Pair*{\Parens*{A_\sigma}}{\Interpretation{}}\) be a \(\Sigma\)-model and \(\rho\) a valuation \(\Parens*{X_\sigma \to A_\sigma}\).
\begin{parts}
\part
Show that
\begin{equation*}
\Interpretation*{M\Substitution*{N/x}}_\rho
= \Interpretation*{M}_{\rho(x \mapsto \Interpretation*{N}_\rho)}.
\end{equation*}
\part
Let \(\Pair*{\Parens*{B_\sigma}}{\Interpretation{}}\) be another \(\Sigma\)-model and \(H_\sigma : A_\sigma \to B_\sigma\) be a morphism of \(\Sigma\)-models (\emph{i.e.} a morphism in the fibre of \(\SModel\) over \(\Sigma\)).
Show that
\begin{equation*}
H\Parens*{\Interpretation*{M}_\rho}
= \Interpretation*{M}_{H \circ \rho}.
\end{equation*}
\end{parts}
\end{exercise}

\begin{partsolution}{i}
Let \(T = \Carrier{\Sigma}\) be the set of types. The valuation is the family \(\rho = \IndexedFamily*{\sigma \in T}{\rho_\sigma : X_\sigma \to A_\sigma}\).
We proceed by induction on terms.
First, for \(\tau\in T\), \(x, y \in X_\tau\), and \(N \in \Terms_\tau(X)\),
\begin{align*}
\Interpretation*{y\Substitution*{N/x}}_\rho^\tau
&=\begin{cases}
\Interpretation*{N}_\rho^\tau & \text{if \(y = x\)} \\
\Interpretation*{y}_\rho^\tau & \text{otherwise}
\end{cases} \\
&=\begin{cases}
\Interpretation*{N}_\rho^\tau & \text{if \(y = x\)} \\
\rho_\tau(y) & \text{otherwise}
\end{cases} \\
&= \rho_\tau(x \mapsto \Interpretation*{N}_\rho^\tau)(y) \\
&=\Interpretation*{y}_{\rho(x \mapsto \Interpretation*{N}_\rho^\tau)}^\tau.
\end{align*}
Next, if \(\tau_0,\ldots,\tau_n\in T\), \(M_0 \in \Terms_{\tau_0}(X), \ldots, M_{n-1} \in \Terms_{\tau_{n-1}}(X)\), and \(F : \tau_0,\,\ldots,\,\tau_{n-1} \to \tau_n\) is a function symbol in \(\Sigma\), then, by induction,
\begin{align*}
&\Interpretation*{F\Parens{M_0,\ldots,M_{n-1}}\Substitution*{N/x}}_\rho^{\tau_n} \\
&\qquad=\Interpretation*{F\Parens{M_0\Substitution*{N/x},\ldots,M_{n-1}\Substitution*{N/x}}}_\rho^{\tau_n} \\
&\qquad=\Interpretation*{F}\Parens*{\Interpretation*{M_0\Substitution*{N/x}}_\rho^{\tau_0},\ldots,\Interpretation*{M_{n-1}\Substitution*{N/x}}_\rho^{\tau_{n-1}}} \\
&\qquad=\Interpretation*{F}\Parens*{\Interpretation*{M_0}_{\rho(x \mapsto \Interpretation*{N}_\rho^\tau)}^{\tau_0},\ldots,\Interpretation*{M_{n-1}}_{\rho(x \mapsto \Interpretation*{N}_\rho^\tau)}^{\tau_{n-1}}} \\
&\qquad=\Interpretation*{F\Parens{M_0,\ldots,M_{n-1}}}_{\rho(x \mapsto \Interpretation*{N}_\rho^\tau)}^{\tau_n}.
\qedhere
\end{align*}
\end{partsolution}

\begin{partsolution}{ii}
Let \(\Pair*{\IndexedFamily{\sigma\in T}{A_\sigma}}{\Interpretation{}_A}\) and \(\Pair*{\IndexedFamily{\sigma\in T}{B_\sigma}}{\Interpretation{}_B}\) be \(\Sigma\)-models, and let \(H = \IndexedFamily*{\sigma \in T}{H_\sigma : A_\sigma \to B_\sigma}\) be a morphism of \(\Sigma\)-models.
If \(\tau\in T\) and \(x \in X_\tau\), then
\begin{equation*}
H_\tau\Parens*{\Interpretation*{x}_{A,\,\rho}^\tau}
=H_\tau\Parens*{\rho_\tau(x)}
=\Interpretation*{x}_{B,\,H \circ \rho}^\tau.
\end{equation*}
Next, if \(\tau_0,\ldots,\tau_n\in T\), \(M_0 \in \Terms_{\tau_0}(X), \ldots, M_{n-1} \in \Terms_{\tau_{n-1}}(X)\), and \(F : \tau_0,\,\ldots,\,\tau_{n-1} \to \tau_n\) is a function symbol in \(\Sigma\), then, by induction,
\begin{align*}
&H_{\tau_n}\Parens*{\Interpretation*{F\Parens*{M_0,\ldots,M_{n-1}}}_{A,\,\rho}^{\tau_n}} \\
&\qquad= H_{\tau_n}\Parens*{\Interpretation*{F}_A\Parens*{\Interpretation*{M_0}_{A,\,\rho}^{\tau_0},\ldots,\Interpretation*{M_{n-1}}_{A,\,\rho}^{\tau_{n-1}}}} \\
&\qquad= \Interpretation*{F}_B\Parens*{H_{\tau_0}\Parens*{\Interpretation*{M_0}_{A,\,\rho}^{\tau_0}},\ldots,H_{\tau_{n-1}}\Parens*{\Interpretation*{M_{n-1}}_{A,\,\rho}^{\tau_{n-1}}}} \\
&\qquad= \Interpretation*{F}_B\Parens*{\Interpretation*{M_0}_{B,\,H\circ\rho}^{\tau_0},\ldots,\Interpretation*{M_{n-1}}_{B,\,H\circ\rho}^{\tau_{n-1}}} \\
&\qquad=\Interpretation*{F\Parens*{M_0,\ldots,M_{n-1}}}_{B,\,H \circ \rho}^{\tau_n}.
\qedhere
\end{align*}
\end{partsolution}
