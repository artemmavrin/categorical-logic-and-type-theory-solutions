\begin{exercise}
Prove that a morphism \(\Pair*{u}{\IndexedFamily{i \in I}{f_i}}\) in \(\Fam(\cat{C})\) is Cartesian if and only if each \(f_i\) is an isomorphism in \(\cat{C}\).
\end{exercise}

\begin{solution}
(\(\Longrightarrow\))
Suppose
\begin{equation*}
\Pair*{u}{\IndexedFamily{i \in I}{f_i}} : \IndexedFamily{i \in I}{X_i} \to \IndexedFamily{j \in J}{Y_j}
\end{equation*}
is a Cartesian morphism in \(\Fam(\cat{C})\) over a morphism \(u : I \to J\) in \(\Sets\).
We also know that
\begin{equation*}
\Pair*{u}{\IndexedFamily*{i \in I}{\id{Y_{u(i)}}}} : \IndexedFamily*{i \in I}{Y_{u(i)}} \to \IndexedFamily{j \in J}{Y_j}
\end{equation*}
is another Cartesian lift of \(u\) (cf. \cite[preceding Definition~1.2.1]{MR1674451}).
Thus, by \expartref{1}{1}{1}{i}, there is a unique vertical isomorphism
\begin{equation*}
\Pair*{\id{I}}{\IndexedFamily*{i \in I}{X_i \overset{g_i}{\to} Y_{u(i)}}} : \IndexedFamily{i \in I}{X_i} \to \IndexedFamily*{i \in I}{Y_{u(i)}}
\end{equation*}
such that
\begin{equation*}
\Pair*{u}{\IndexedFamily{i \in I}{f_i}}
= \Pair*{u}{\IndexedFamily*{i \in I}{\id{Y_{u(i)}}}} \circ \Pair*{\id{I}}{\IndexedFamily{i \in I}{g_i}}
\end{equation*}
in \(\Fam(\cat{C})\).
In particular, \(f_i = \id{Y_{u(i)}} \circ g_i = g_i\) for all \(i \in I\), so each \(f_i\) is an isomorphism.

(\(\Longleftarrow\))
Conversely, suppose
\begin{equation*}
\Pair*{u}{\IndexedFamily{i \in I}{f_i}} : \IndexedFamily{i \in I}{X_i} \to \IndexedFamily{j \in J}{Y_j}
\end{equation*}
is a morphism in \(\Fam(\cat{C})\) over a morphism \(u : I \to J\) in \(\Sets\) such that each \(f_i : X_i \to Y_{u(i)}\) is an isomorphism.
We claim that \(\Pair*{u}{\IndexedFamily{i \in I}{f_i}}\) is Cartesian.
Suppose \(v : K \to I\) is another morphism in \(\Sets\) and
\begin{equation*}
\Pair*{u \circ v}{\IndexedFamily{k \in K}{g_k}} : \IndexedFamily{k \in K}{Z_k} \to \IndexedFamily{j \in J}{Y_j}
\end{equation*}
is a morphism in \(\Fam(\cat{C})\) over \(u \circ v\), as in the following diagram.
\begin{equation*}
\begin{tikzcd}[row sep=small]
\IndexedFamily{k \in K}{Z_k} \arrow[rrd, bend left, "{\Pair*{u \circ v}{\IndexedFamily{k \in K}{g_k}}}"] \arrow[dr, dashed, "?"] \arrow[dd, -Triangle] \\
& \IndexedFamily{i \in I}{X_i} \arrow[r, "{\Pair*{u}{\IndexedFamily{i \in I}{f_i}}}"] \arrow[dd, -Triangle]
& \IndexedFamily{j \in J}{Y_j} \arrow[dd, -Triangle] \\
K \arrow[dr, "v"{below left}]\\
& I \arrow[r, "u"]
& J
\end{tikzcd}
\end{equation*}
Then the unique filler of the dashed ``\(?\)'' morphism in this diagram is \(? = \Pair*{v}{\IndexedFamily*{k \in K}{f_{v(k)}^{-1} \circ g_k}}\), so \(\Pair*{u}{\IndexedFamily{i \in I}{f_i}}\) is Cartesian.
\end{solution}
