\begin{exercise}
Define a left adjoint \(F : \Fam(\Sets) \to \Pred\) to the inclusion functor
\begin{equation*}
\begin{tikzcd}[column sep=small]
\Pred \arrow[dr] \arrow[rr, hookrightarrow] && \Fam(\Sets) \arrow[dl] \arrow[ll, bend right, "F"{above}] \\
& \Sets
\end{tikzcd}
\end{equation*}
such that: (1) \(F\) makes the triangle commute (so it does not change the index set), and (2) \(F\) commutes with substitution.
\end{exercise}

\begin{solution}
Recall that the inclusion functor \(G : \Pred \hookrightarrow \Fam(\Sets)\) sends a predicate \(\Pair{I}{X}\) (where \(X \subseteq I\)) to the family \(G\Parens{I, X} = \IndexedFamily{i \in I}{X_i}\), where
\begin{equation*}
X_i =
\begin{cases}
\SetOf{*} & \text{if \(i \in X\)} \\
\emptyset & \text{otherwise.}
\end{cases}
\end{equation*}
Define the functor \(F : \Fam(\Sets) \to \Pred\) on objects by
\begin{equation*}
F\Parens*{\IndexedFamily{i \in I}{X_i}}
= \Pair*{I}{\SetOf{i \in I \SuchThat X_i \neq \emptyset}}
\end{equation*}
and on morphisms by
\begin{equation*}
F\Parens*{\IndexedFamily{i \in I}{X_i} \xrightarrow{\Pair{u}{f}} \IndexedFamily{j \in J}{Y_j}}
= u.
\end{equation*}
Note that \(F\) is well-defined on morphisms.
Indeed, suppose \(\IndexedFamily{i \in I}{X_i}\) and \(\IndexedFamily{j \in J}{Y_j}\) are families of sets and \(\Pair{u}{f} : \IndexedFamily{i \in I}{X_i} \to \IndexedFamily{j \in J}{Y_j}\) is a morphism of families.
If \(i \in I\) and \(X_i \neq \emptyset\), then \(\emptyset \neq f_i(X_i) \subseteq Y_{u(i)}\), so \(Y_{u(i)} \neq \emptyset\), and hence \(u\) is a morphism from \(F\Parens*{\IndexedFamily{i \in I}{X_i}}\) to \(F\Parens*{\IndexedFamily{j \in J}{Y_j}}\) in \(\Pred\).
Moreover, \(F\) does not change the index set by definition.

Next, we claim that \(F \dashv G\) by demonstrating the adjointness criteria in \cite[Definition~3.1.4]{MR1291599}, namely that there is a ``unit'' natural transformation
\begin{equation*}
\eta : \id{\Fam(\Sets)} \to G \circ F.
\end{equation*}
such that for each \(I\)-indexed family of sets \(\IndexedFamily{i \in I}{X_i}\), each object \(\Pair{J}{Y}\) in \(\Pred\), and each morphism \(\Pair{v}{g} : \IndexedFamily{i \in I}{X_i} \to G\Parens{J, Y}\) in \(\Fam(\Sets)\), there exists a unique morphism \(v^\prime : F\Parens*{\IndexedFamily{i \in I}{X_i}} \to \Pair{J}{Y}\) in \(\Pred\) such that \(G(v^\prime) \circ \eta_{\IndexedFamily{i \in I}{X_i}} = \Pair{v}{g}\), as in the following commutative diagram.
\begin{equation*}
\begin{tikzcd}[column sep=small]
\Pred &&& \Fam(\Sets) \\[-5ex]
F\Parens*{\IndexedFamily{i \in I}{X_i}} \arrow[d, dashed, "v^\prime"]
&& \IndexedFamily{i \in I}{X_i} \arrow[rr, "\eta_{\IndexedFamily{i \in I}{X_i}}"] \arrow[drr, "{\Pair{v}{g}}"{below left}] && G\Parens*{F\Parens*{\IndexedFamily{i \in I}{X_i}}} \arrow[d, dashed, "G(v^\prime)"]
\\
\Pair{J}{Y}
&&&& G\Parens{J, Y}
\end{tikzcd}
\end{equation*}
To this end, for an object \(\IndexedFamily{i \in I}{X_i}\) define the component
\begin{equation*}
\eta_{\IndexedFamily{i \in I}{X_i}} : \IndexedFamily{i \in I}{X_i} \to G\Parens*{F\Parens*{\IndexedFamily{i \in I}{X_i}}}
= \IndexedFamily[\big]{i \in I}{\ite{X_i \neq \emptyset}{\SetOf{*}}{\emptyset}}
\end{equation*}
as
\begin{equation*}
\eta_{\IndexedFamily{i \in I}{X_i}} = \Pair*{\id{I}}{\IndexedFamily[\big]{i \in I}{f_i : X_i \to \ite{X_i \neq \emptyset}{\SetOf{*}}{\emptyset}}},
\end{equation*}
where
\begin{equation*}
f_i = \begin{cases}
x \mapsto * & \text{if \(X_i \neq \emptyset\)} \\
\id{\emptyset} & \text{otherwise.}
\end{cases}
\end{equation*}
We omit the rest of the verification that \(\eta\) is the unit of the adjunction \(F \dashv G\).

Finally, we claim that \(F\) commutes with substitution.
Indeed, if \(u : I \to J\) is a morphism in \(\Sets\) and \(\IndexedFamily{j \in J}{Y_j}\) is a \(J\)-indexed family of sets, then
\begin{align*}
u^*\Parens*{F\Parens*{\IndexedFamily{j \in J}{Y_j}}}
&= u^*\Pair*{J}{\SetOf{j \in J \SuchThat Y_j \neq \emptyset}} \\
&= \Pair*{I}{\SetOf[\big]{i \in I \SuchThat Y_{u(i)} \neq \emptyset}} \\
&= F\Parens*{\IndexedFamily*{i \in I}{Y_{u(i)}}} \\
&= F\Parens*{u^*\Parens*{\IndexedFamily{j \in J}{Y_j}}}.
\qedhere
\end{align*}
\end{solution}
