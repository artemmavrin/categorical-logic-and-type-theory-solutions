\begin{exercise}
Show that the equality functor \(\Eq : \Sets \to \Rel{}\) also has a right adjoint.
\end{exercise}

\begin{solution}
Define the functor
\begin{equation*}
F : \Rel{} \to \Sets
\end{equation*}
on objects by
\begin{equation*}
F\Parens{I, R} = \SetOf{i \in I \SuchThat \Pair{i}{i} \in R}
\end{equation*}
and on morphisms by
\begin{equation*}
F\Parens*{\Pair{I}{R} \overset{u}{\to} \Pair{J}{S}}
= u|_{F\Parens{I, R}}.
\end{equation*}
Note that \(F\) is well-defined on morphisms since if \(u : \Pair{I}{R} \to \Pair{J}{S}\) is a morphism in \(\Rel{}\) and \(i \in F\Parens{I, R}\), then \(\Pair{i}{i} \in R\), so \(\Pair{u(i)}{u(i)} \in S\), and hence \(u(i) \in F\Parens{J, S}\).
We claim that \(\Eq \dashv F\).
Fix an object \(I\) in \(\Sets\) and an object \(\Pair{J}{S}\) in \(\Rel{}\).
Define functions
\begin{align*}
f &: \Hom{\Rel{}}{\Eq(I)}{\Pair{J}{S}} \to \Hom{\Sets}{I}{F\Parens{J, S}} \\
g &: \Hom{\Sets}{I}{F\Parens{J, S}} \to \Hom{\Rel{}}{\Eq(I)}{\Pair{J}{S}}
\end{align*}
by
\begin{align*}
f\Parens*{\Eq(I) \overset{u}{\to} \Pair{J}{S}}
&= u &
g\Parens*{I \overset{u}{\to} F\Parens{J, S}}
&= u.
\end{align*}
Note that \(f\) is well-defined since if \(u : \Eq(I) \to \Pair{J}{S}\) is a morphism in \(\Rel{}\) and \(i \in I\), then \(\Pair{i}{i} \in \Eq(I)\), so \(\Pair{u(i)}{u(i)} \in S\), and hence \(u(i) \in F\Parens{J, S}\).
Similarly, note that \(g\) is well-defined since if \(u : I \to F\Parens{J, S}\) is a morphism in \(\Sets\) and \(i \in I\), then \(u(i) \in F\Parens{J, S}\), so \(\Pair{u(i)}{u(i)} \in S\).
Thus, \(f\) and \(g\) are mutual inverses, so we have
\begin{equation*}
\Hom{\Rel{}}{\Eq(I)}{\Pair{J}{S}} \cong \Hom{\Sets}{I}{F\Parens{J, S}}.
\end{equation*}
We omit the proof of naturality of this bijection and conclude that \(\Eq \dashv F\).
\end{solution}
